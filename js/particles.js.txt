/**
 * Движок фоновых частиц для Аниме-Архива
 * Поддерживает 20 пресетов, настройку плотности, скорости и размера.
 */

class ParticleSystem {
    constructor() {
        this.canvas = document.getElementById('particles-canvas');
        this.ctx = this.canvas.getContext('2d');
        this.particles = [];
        this.animationId = null;
        this.isEnabled = true;
        
        // Настройки по умолчанию
        this.config = {
            preset: 'sakura',
            density: 50,
            speed: 3,
            size: 8,
            theme: 'dark'
        };

        this.resize();
        window.addEventListener('resize', () => this.resize());
        this.loadConfig();
        this.init();
    }

    resize() {
        this.canvas.width = window.innerWidth;
        this.canvas.height = window.innerHeight;
    }

    loadConfig() {
        const saved = localStorage.getItem('animeArchive_settings');
        if (saved) {
            const settings = JSON.parse(saved);
            this.config = { ...this.config, ...settings.particles };
            this.isEnabled = settings.particles?.enabled !== false;
        }
    }

    updateConfig(newConfig) {
        this.config = { ...this.config, ...newConfig };
        this.init(); // Пересоздаём частицы при изменении настроек
    }

    getPresetColors(preset) {
        const colors = {
            'sakura': ['#ffb7c5', '#ffc0cb', '#ff9eb5'],
            'white-sakura': ['#ffffff', '#f0f0f0', '#e0e0e0'],
            'red-roses': ['#ff0040', '#cc0033', '#990026'],
            'white-petals': ['#ffffff', '#f5f5f5', '#e8e8e8'],
            'yellow-petals': ['#ffd700', '#ffec8b', '#fffacd'],
            'snow': ['#ffffff', '#e6f3ff', '#cce7ff'],
            'rain': ['#a0c4ff', '#7ab8ff', '#55a3ff'],
            'drizzle': ['#b3d9ff', '#99ccff', '#80bfff'],
            'autumn': ['#ff8c00', '#ff4500', '#daa520', '#8b4513'],
            'fireflies': ['#ffff00', '#adff2f', '#00ff00'],
            'shooting-stars': ['#ffffff', '#fffacd', '#e6e6fa'],
            'stardust': ['#e0ffff', '#00ffff', '#87cefa'],
            'bubbles': ['rgba(255,255,255,0.3)', 'rgba(255,255,255,0.5)', 'rgba(255,255,255,0.1)'],
            'confetti': ['#ff0000', '#00ff00', '#0000ff', '#ffff00', '#ff00ff'],
            'feathers': ['#ffffff', '#f5f5dc', '#e6e6fa'],
            'hearts': ['#ff1493', '#ff69b4', '#ffb6c1'],
            'butterflies': ['#ff8c00', '#ffd700', '#ff6347'],
            'clover': ['#32cd32', '#228b22', '#006400'],
            'sparks': ['#ffa500', '#ff4500', '#ffff00'],
            'night-petals': ['#191970', '#4169e1', '#000080']
        };
        return colors[preset] || colors['sakura'];
    }

    createParticle() {
        const colors = this.getPresetColors(this.config.preset);
        const isRain = this.config.preset === 'rain' || this.config.preset === 'drizzle';
        const isStar = this.config.preset === 'shooting-stars';
        
        return {
            x: Math.random() * this.canvas.width,
            y: Math.random() * this.canvas.height - this.canvas.height,
            size: (Math.random() * 0.5 + 0.5) * this.config.size,
            speedX: (Math.random() - 0.5) * (this.config.preset === 'rain' ? 1 : 2),
            speedY: (Math.random() * 1 + 0.5) * this.config.speed * (isRain ? 3 : 1),
            color: colors[Math.floor(Math.random() * colors.length)],
            rotation: Math.random() * 360,
            rotationSpeed: (Math.random() - 0.5) * 2,
            opacity: Math.random() * 0.5 + 0.5,
            isStar: isStar,
            trail: isStar ? [] : null
        };
    }

    init() {
        this.particles = [];
        if (!this.isEnabled) {
            this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
            if (this.animationId) cancelAnimationFrame(this.animationId);
            return;
        }

        const count = this.config.density;
        for (let i = 0; i < count; i++) {
            this.particles.push(this.createParticle());
        }

        if (!this.animationId) {
            this.animate();
        }
    }

    animate() {
        if (!this.isEnabled) return;

        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

        this.particles.forEach(p => {
            this.ctx.save();
            this.ctx.globalAlpha = p.opacity;
            this.ctx.fillStyle = p.color;
            this.ctx.strokeStyle = p.color;

            if (p.isStar) {
                // Падающие звёзды с хвостом
                p.trail.push({ x: p.x, y: p.y });
                if (p.trail.length > 10) p.trail.shift();
                
                this.ctx.beginPath();
                this.ctx.moveTo(p.trail[0].x, p.trail[0].y);
                for (let i = 1; i < p.trail.length; i++) {
                    this.ctx.lineTo(p.trail[i].x, p.trail[i].y);
                }
                this.ctx.lineWidth = p.size * 0.5;
                this.ctx.stroke();

                p.x += p.speedX * 3;
                p.y += p.speedY * 3;
            } else if (this.config.preset === 'rain' || this.config.preset === 'drizzle') {
                // Дождь
                this.ctx.beginPath();
                this.ctx.moveTo(p.x, p.y);
                this.ctx.lineTo(p.x + p.speedX, p.y + p.size * 2);
                this.ctx.lineWidth = p.size * 0.3;
                this.ctx.stroke();
                
                p.x += p.speedX;
                p.y += p.speedY;
            } else {
                // Обычные частицы (лепестки, снег и т.д.)
                this.ctx.translate(p.x, p.y);
                this.ctx.rotate((p.rotation * Math.PI) / 180);
                
                if (this.config.preset === 'hearts') {
                    this.drawHeart(0, 0, p.size);
                } else if (this.config.preset === 'bubbles') {
                    this.ctx.beginPath();
                    this.ctx.arc(0, 0, p.size, 0, Math.PI * 2);
                    this.ctx.fill();
                } else {
                    // Эллипс для лепестков/листьев
                    this.ctx.beginPath();
                    this.ctx.ellipse(0, 0, p.size, p.size * 0.6, 0, 0, Math.PI * 2);
                    this.ctx.fill();
                }

                p.x += p.speedX + Math.sin(p.y * 0.01) * 0.5; // Лёгкое покачивание
                p.y += p.speedY;
                p.rotation += p.rotationSpeed;
            }

            this.ctx.restore();

            // Сброс частицы при выходе за границы
            if (p.y > this.canvas.height + 20 || p.x > this.canvas.width + 20 || p.x < -20) {
                Object.assign(p, this.createParticle());
                p.y = -20; // Начинаем сверху
            }
        });

        this.animationId = requestAnimationFrame(() => this.animate());
    }

    drawHeart(x, y, size) {
        this.ctx.beginPath();
        const topCurveHeight = size * 0.3;
        this.ctx.moveTo(x, y + topCurveHeight);
        this.ctx.bezierCurveTo(x, y, x - size / 2, y, x - size / 2, y + topCurveHeight);
        this.ctx.bezierCurveTo(x - size / 2, y + (size + topCurveHeight) / 2, x, y + (size + topCurveHeight) / 2, x, y + size);
        this.ctx.bezierCurveTo(x, y + (size + topCurveHeight) / 2, x + size / 2, y + (size + topCurveHeight) / 2, x + size / 2, y + topCurveHeight);
        this.ctx.bezierCurveTo(x + size / 2, y, x, y, x, y + topCurveHeight);
        this.ctx.fill();
    }

    toggle(enabled) {
        this.isEnabled = enabled;
        if (enabled) {
            this.init();
        } else {
            this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
            if (this.animationId) {
                cancelAnimationFrame(this.animationId);
                this.animationId = null;
            }
        }
    }
}

// Экспортируем глобально для использования в app.js
window.particleSystem = new ParticleSystem();
